# libp2pml: QuickJS and the engine. The Android build (AGP's externalNativeBuild)
# makes the shared library for each ABI. The host build (P2PML_HOST, from the
# Gradle task hostNativeBuild) makes the same library for the JVM host tests,
# and the C tests with sanitizers.

cmake_minimum_required(VERSION 3.22)
project(p2pml C CXX)

set(CMAKE_C_STANDARD 11)
# gnu11, not c11: the code uses POSIX and platform functions, such as strdup.
set(CMAKE_C_EXTENSIONS ON)
# Only the JNI functions are exported, which also lets the linker drop the
# parts of QuickJS the engine does not use.
set(CMAKE_C_VISIBILITY_PRESET hidden)

option(P2PML_HOST "Build for the host: the JNI library for JVM tests and the C tests" OFF)
set(P2PML_CLANG_TIDY "" CACHE FILEPATH "clang-tidy to run on our sources, every finding an error")

find_package(Threads REQUIRED)

set(QUICKJS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/third_party/quickjs)
file(STRINGS ${QUICKJS_DIR}/VERSION QUICKJS_VERSION)

# QuickJS as released, with the defines and code flags of its own Makefile.
# The optimization follows the build type: a debug build runs it slower.
function(add_quickjs name)
  add_library(${name} STATIC
    ${QUICKJS_DIR}/quickjs.c
    ${QUICKJS_DIR}/libregexp.c
    ${QUICKJS_DIR}/libunicode.c
    ${QUICKJS_DIR}/cutils.c
    ${QUICKJS_DIR}/dtoa.c)
  target_compile_definitions(${name} PRIVATE _GNU_SOURCE CONFIG_VERSION="${QUICKJS_VERSION}")
  target_compile_options(${name} PRIVATE -fwrapv -funsigned-char -w)
  set_target_properties(${name} PROPERTIES POSITION_INDEPENDENT_CODE ON)
  # SYSTEM: our warning flags do not apply to its headers.
  target_include_directories(${name} SYSTEM PUBLIC ${QUICKJS_DIR})
  target_link_libraries(${name} PUBLIC Threads::Threads)
  if(NOT APPLE)
    target_link_libraries(${name} PUBLIC m)
  endif()
endfunction()

function(use_our_checks target)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  if(P2PML_CLANG_TIDY)
    set_target_properties(${target} PROPERTIES
      C_CLANG_TIDY "${P2PML_CLANG_TIDY};--warnings-as-errors=*;--quiet")
  endif()
endfunction()

# Builds JS files into a C file of p2pml_script entries (src/polyfills.h), in
# the order given: `const p2pml_script <list>[]` and `<list>_count`, each
# source with a NUL after it, as JS_Eval needs. The files are configure
# dependencies, so an edit generates the C file again; it is written only when
# its content changes.
function(embed_js output list)
  set(arrays "")
  set(entries "")
  set(count 0)
  foreach(path IN LISTS ARGN)
    file(READ ${path} hex HEX)
    string(REGEX REPLACE "([0-9a-f][0-9a-f])" "0x\\1," bytes "${hex}")
    get_filename_component(name ${path} NAME)
    string(APPEND arrays "static const unsigned char source_${count}[] = {${bytes}0};\n")
    string(APPEND entries "    {\"${name}\", source_${count}},\n")
    math(EXPR count "${count} + 1")
  endforeach()
  set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS ${ARGN})
  file(WRITE ${output}.new
    "// Generated by native/CMakeLists.txt (embed_js) from JS files: do not edit.\n\n"
    "#include \"polyfills.h\"\n\n${arrays}\n"
    "const p2pml_script ${list}[] = {\n${entries}};\n"
    "const int ${list}_count = ${count};\n")
  configure_file(${output}.new ${output} COPYONLY)
endfunction()

# The JS polyfills, in the order the engine evaluates them.
set(POLYFILLS webidl.js dom_exception.js events.js encoding.js base64.js url.js fetch.js websocket.js)
list(TRANSFORM POLYFILLS PREPEND ${CMAKE_CURRENT_SOURCE_DIR}/src/polyfills/)
embed_js(${CMAKE_BINARY_DIR}/polyfills.c p2pml_polyfills ${POLYFILLS})

# The web core's bundle, the module p2p-media-loader-core (src/modules.c): from
# the npm package that package.json pins. Gradle installs it (:native:npmCi)
# before it runs CMake; Android Studio's sync runs CMake without that task.
set(CORE_BUNDLE
  ${CMAKE_CURRENT_SOURCE_DIR}/node_modules/p2p-media-loader-core/dist/p2p-media-loader-core-hls.es.js)
if(NOT EXISTS ${CORE_BUNDLE})
  message(FATAL_ERROR "${CORE_BUNDLE} is missing: run ./gradlew :native:npmCi")
endif()
embed_js(${CMAKE_BINARY_DIR}/core_bundle.c p2pml_core_bundle ${CORE_BUNDLE})

set(ENGINE_SOURCES src/engine.c src/encoding.c src/url.c src/modules.c src/channel.c src/http.c src/websocket.c
  ${CMAKE_BINARY_DIR}/polyfills.c ${CMAKE_BINARY_DIR}/core_bundle.c)

# Ada as released (third_party/ada): C++20, without exceptions, RTTI and
# URLPattern (so no std::regex). Our C calls only its C API, ada_c.h.
function(add_ada name)
  add_library(${name} STATIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ada/ada.cpp)
  set_target_properties(${name} PROPERTIES
    CXX_STANDARD 20
    CXX_STANDARD_REQUIRED ON
    CXX_EXTENSIONS OFF
    POSITION_INDEPENDENT_CODE ON
    # Not exported from libp2pml.so, so that the linker drops what is unused.
    CXX_VISIBILITY_PRESET hidden
    VISIBILITY_INLINES_HIDDEN ON)
  target_compile_definitions(${name} PRIVATE ADA_INCLUDE_URL_PATTERN=0)
  target_compile_options(${name} PRIVATE -fno-exceptions -fno-rtti -w)
  target_include_directories(${name} SYSTEM PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/ada)
endfunction()

add_quickjs(quickjs)
add_ada(ada)
# Ada turns on its development checks in a build without NDEBUG, such as an
# Android debug build; a failed check aborts the process. Only the C tests
# keep them (ada_sanitized).
target_compile_definitions(ada PRIVATE ADA_DEVELOPMENT_CHECKS=0)

add_library(p2pml SHARED ${ENGINE_SOURCES} src/jni.c)
target_include_directories(p2pml PRIVATE src)
target_link_libraries(p2pml PRIVATE quickjs ada)
if(ANDROID)
  # The static libraries' symbols (libc++, which Ada needs, among them) are
  # not exported: only the JNI functions are.
  target_link_options(p2pml PRIVATE "LINKER:--exclude-libs,ALL")
endif()
use_our_checks(p2pml)

if(P2PML_HOST)
  # The NDK's jni.h, which needs only the C library's headers: a JDK such as
  # Android Studio's may have no JNI headers. Copied alone, so that the rest
  # of the Android sysroot stays out of the host build.
  set(P2PML_JNI_H "" CACHE FILEPATH "jni.h for the host build")
  if(NOT EXISTS "${P2PML_JNI_H}")
    message(FATAL_ERROR "P2PML_JNI_H must name a jni.h, such as the NDK's")
  endif()
  configure_file(${P2PML_JNI_H} ${CMAKE_BINARY_DIR}/jni/jni.h COPYONLY)
  target_include_directories(p2pml SYSTEM PRIVATE ${CMAKE_BINARY_DIR}/jni)

  # The C tests: the engine with AddressSanitizer and UndefinedBehaviorSanitizer,
  # QuickJS included. Asserts stay on, so JS_FreeRuntime checks that no JS
  # object leaked.
  set(SANITIZERS -fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_quickjs(quickjs_sanitized)
  # -O1: at -O0 with ASan, a JS call takes so much of the C stack that
  # QuickJS's limit allows only about 10 nested calls (353 at -O1).
  target_compile_options(quickjs_sanitized PRIVATE ${SANITIZERS} -O1)
  add_ada(ada_sanitized)
  target_compile_options(ada_sanitized PRIVATE ${SANITIZERS})
  # The JS tests that engine_test runs, each after the setup scripts: the
  # harness, then the test data.
  file(GLOB JS_TEST_DATA CONFIGURE_DEPENDS
    ${CMAKE_CURRENT_SOURCE_DIR}/test/third_party/wpt-url/*_data.js)
  embed_js(${CMAKE_BINARY_DIR}/test_setup.c p2pml_test_setup
    ${CMAKE_CURRENT_SOURCE_DIR}/test/js/harness.js ${JS_TEST_DATA})
  file(GLOB JS_TESTS CONFIGURE_DEPENDS ${CMAKE_CURRENT_SOURCE_DIR}/test/js/*_test.js)
  embed_js(${CMAKE_BINARY_DIR}/test_scripts.c p2pml_test_scripts ${JS_TESTS})
  add_executable(engine_test test/engine_test.c ${ENGINE_SOURCES}
    ${CMAKE_BINARY_DIR}/test_setup.c ${CMAKE_BINARY_DIR}/test_scripts.c)
  target_include_directories(engine_test PRIVATE src)
  target_compile_options(engine_test PRIVATE ${SANITIZERS})
  target_link_options(engine_test PRIVATE ${SANITIZERS})
  target_link_libraries(engine_test PRIVATE quickjs_sanitized ada_sanitized)
  use_our_checks(engine_test)

  enable_testing()
  add_test(NAME engine COMMAND engine_test)
  # A broken loop hangs instead of failing.
  set_tests_properties(engine PROPERTIES TIMEOUT 60)
endif()
